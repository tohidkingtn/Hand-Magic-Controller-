async function sendMessage() { const apiKey = localStorage.getItem('gemini_api_key') || document.getElementById('api-key-input').value.trim(); if (!apiKey) { alert("Pehle Upar 'Key Settings' me jaakar apni API Key Save Karein!"); document.getElementById('key-section').classList.remove('hidden'); return; } const inputEl = document.getElementById('user-input'); const chatBox = document.getElementById('chat-box'); const prompt = inputEl.value.trim(); if (!prompt) return; chatBox.innerHTML += `
${prompt}
`; inputEl.value = ''; chatBox.scrollTop = chatBox.scrollHeight; const loadingId = "loading-" + Date.now(); chatBox.innerHTML += `
Soch raha hoon...
`; chatBox.scrollTop = chatBox.scrollHeight; try { // gemini-1.5-flash-latest model endpoint const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash-latest:generateContent`, { method: "POST", headers: { "Content-Type": "application/json", "x-goog-api-key": apiKey }, body: JSON.stringify({ system_instruction: { parts: [{ text: systemPrompt }] }, contents: [{ parts: [{ text: prompt }] }] }) }); const data = await response.json(); if (!response.ok || data.error) { const errorMsg = data.error?.message || "Invalid Key ya Model Error"; document.getElementById(loadingId).outerHTML = `
Google Error: ${errorMsg}
`; return; } const reply = data.candidates[0].content.parts[0].text; document.getElementById(loadingId).outerHTML = `
${reply}
`; } catch (error) { document.getElementById(loadingId).outerHTML = `
Error: ${error.message}
`; } chatBox.scrollTop = chatBox.scrollHeight; }
